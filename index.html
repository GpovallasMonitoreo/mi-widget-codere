module.exports = async function (req, res) { // 1. Cabeceras CORS abiertas res.setHeader('Access-Control-Allow-Credentials', true); res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET,OPTIONS'); res.setHeader('Access-Control-Allow-Headers', '*'); if (req.method === 'OPTIONS') { return res.status(200).end(); } const endpoint = req.query.endpoint; if (!endpoint) { return res.status(400).json({ error: 'Se requiere un endpoint' }); } try { // CORRECCIÓN DE URL: Quitamos el "/api" que sobraba basado en tu Swagger // Ahora buscará directamente: https://codere-sbs-es.azurewebsites.net/sports const urlCodere = `https://codere-sbs-es.azurewebsites.net${endpoint}`; const response = await fetch(urlCodere, { method: 'GET', headers: { 'Language': 'es-Mx', 'Accept': 'application/json' } }); if (response.status === 204) { return res.status(204).end(); } // SI CODERE RECHAZA LA PETICIÓN, MANDAMOS EL ERROR EXACTO AL FRONTEND if (!response.ok) { const errorText = await response.text(); // Leemos qué dice Codere return res.status(response.status).json({ error: `Codere rechazó la petición con status ${response.status}`, url_intentada: urlCodere, respuesta_codere: errorText }); } const data = await response.json(); return res.status(200).json(data); } catch (error) { // Si Node.js falla antes de contactar a Codere return res.status(500).json({ error: 'Fallo interno en Vercel', detalle: error.toString() }); } };
